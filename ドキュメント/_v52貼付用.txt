[4] 既存クエリ: 通年-記入以外-前処理
----------------------------------------
let
    ソース = let
    folder = if Text.EndsWith(Ptuunen, "\") then Ptuunen else Ptuunen & "\",
    result = fxファイル前処理(folder)
in
    result,
    J判定メーカー = Table.NestedJoin(ソース, {"メーカーマスタ.メーカー名"}, #"セール返答表-判定用-メーカー", {"メーカー名"}, "判定-メーカー", JoinKind.LeftOuter),
    E判定メーカー = Table.ExpandTableColumn(J判定メーカー, "判定-メーカー", {"最新"}, {"判定-メーカー.最新"}),
    J判定親カテ = Table.NestedJoin(E判定メーカー, {"メーカーマスタ.メーカー名", "親カテゴリー"}, #"セール返答表-判定用-親カテ", {"メーカー名", "親カテ"}, "判定-親カテ", JoinKind.LeftOuter),
    E判定親カテ = Table.ExpandTableColumn(J判定親カテ, "判定-親カテ", {"最新"}, {"判定-親カテ.最新"}),
    J判定子カテ = Table.NestedJoin(E判定親カテ, {"メーカーマスタ.メーカー名", "ブランド"}, #"セール返答表-判定用-子カテ", {"メーカー名", "子カテ"}, "判定-子カテ", JoinKind.LeftOuter),
    E判定子カテ = Table.ExpandTableColumn(J判定子カテ, "判定-子カテ", {"最新"}, {"判定-子カテ.最新"}),
    #"名前が変更された列 " = Table.AddColumn(E判定子カテ, "処理方法", each
        if [#"判定-子カテ.最新"] <> null then [#"判定-子カテ.最新"]
        else if [#"判定-親カテ.最新"] <> null then [#"判定-親カテ.最新"]
        else [#"判定-メーカー.最新"]),
    フィルターされた行 = Table.SelectRows(#"名前が変更された列 ", each ([処理方法] = "記入以外前回と同じ")),
    #"トリム テキスト" = Table.TransformColumns(フィルターされた行,{{"ブランド品番", Text.Trim, type text}}),
    除去されたテキスト = Table.TransformColumns(#"トリム テキスト",{{"ブランド品番", Text.Clean, type text}}),
    変更された型 = Table.TransformColumnTypes(除去されたテキスト,{{"元上代", type number}, {"オフ率#(lf)入力(↓)優先", type number}, {"税抜価格#(lf)(設定価格)", type number}, {"税込価格から設定#(lf)入力(↓)", type number}, {"値引き金額#(lf)入力(↓)", type number}, {"税込表示", type number}, {"1%処理-税別", type number}, {"1%処理-税込", type number}})
in
    変更された型

[5] 既存クエリ: 通年-返信あり-前処理
----------------------------------------
let
    ソース = let
    folder = if Text.EndsWith(Ptuunen, "\") then Ptuunen else Ptuunen & "\",
    result = fxファイル前処理(folder)
in
    result,
    変更された型 = Table.TransformColumnTypes(ソース,{{"元上代", type number}, {"オフ率#(lf)入力(↓)優先", type number}, {"値引き金額#(lf)入力(↓)", type number}, {"税込価格から設定#(lf)入力(↓)", type number}, {"税抜価格#(lf)(設定価格)", type number}, {"税込表示", type number}, {"1%処理-税別", type number}, {"1%処理-税込", type number}}),
    J判定メーカー = Table.NestedJoin(変更された型, {"メーカーマスタ.メーカー名"}, #"セール返答表-判定用-メーカー", {"メーカー名"}, "判定-メーカー", JoinKind.LeftOuter),
    E判定メーカー = Table.ExpandTableColumn(J判定メーカー, "判定-メーカー", {"最新"}, {"判定-メーカー.最新"}),
    J判定親カテ = Table.NestedJoin(E判定メーカー, {"メーカーマスタ.メーカー名", "親カテゴリー"}, #"セール返答表-判定用-親カテ", {"メーカー名", "親カテ"}, "判定-親カテ", JoinKind.LeftOuter),
    E判定親カテ = Table.ExpandTableColumn(J判定親カテ, "判定-親カテ", {"最新"}, {"判定-親カテ.最新"}),
    J判定子カテ = Table.NestedJoin(E判定親カテ, {"メーカーマスタ.メーカー名", "ブランド"}, #"セール返答表-判定用-子カテ", {"メーカー名", "子カテ"}, "判定-子カテ", JoinKind.LeftOuter),
    E判定子カテ = Table.ExpandTableColumn(J判定子カテ, "判定-子カテ", {"最新"}, {"判定-子カテ.最新"}),
    #"名前が変更された列 " = Table.AddColumn(E判定子カテ, "処理方法", each
        if [#"判定-子カテ.最新"] <> null then [#"判定-子カテ.最新"]
        else if [#"判定-親カテ.最新"] <> null then [#"判定-親カテ.最新"]
        else [#"判定-メーカー.最新"]),
    フィルターされた行 = Table.SelectRows(#"名前が変更された列 ", each [処理方法] = "返信あり"),
    // 「元上代」と「税抜価格」が等しければ「削除」、そうでなければ元の「要確認」列の値を保持
    要確認の更新 = Table.ReplaceValue(
        フィルターされた行,
        each [要確認],
        each if [元上代] = [#"税抜価格#(lf)(設定価格)"] then "削除" else [要確認],
        Replacer.ReplaceValue,
        {"要確認"}
    )
in
    要確認の更新

let
    ソース = メーカーマスタ,
    対象メーカー = Table.NestedJoin(
        ソース, {"メーカー名"},
        #"セール返答表-記入以外-メーカー", {"メーカー名"},
        "返答表", JoinKind.Inner),
    親カテ指示を除外 = Table.NestedJoin(
        対象メーカー, {"メーカー名", "親カテゴリ名"},
        #"セール返答表-判定用-親カテ", {"メーカー名", "親カテ"},
        "親カテ行", JoinKind.LeftAnti),
    削除された他の列 = Table.SelectColumns(親カテ指示を除外, {"親カテゴリ名"}),
    削除された重複 = Table.Distinct(削除された他の列)
in
    削除された重複
