    // 1. ブランド品番ごとの出現数をカウントする一時的な表を作成
    カウント用 = Table.Group(並べ替えられた列1, {"ブランド品番"}, {{"数", each Table.RowCount(_), Int64.Type}}),

    // 2. 元の表（並べ替えられた列1）にカウント結果を合体させる

    重複列の追加 = Table.AddColumn(展開, "重複", each if [数] > 1 then "重複" else null),

    // 5. 作業用の「数」列を削除して完了

    並べ替えられた列2 = Table.ReorderColumns(最終結果,{"Source.Name", "親カテゴリ", "ブランド品番", "変更後セール価格", "開始日", "終了日", "通年セール", "要確認", "税込価格から設定#(lf)入力(↓)", "新オフ率", "通年セールメーカー", "重複"})
